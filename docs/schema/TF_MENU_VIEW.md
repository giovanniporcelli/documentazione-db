                 UPDATE_TIME,

                 IS_CUSTOM,
                 INSERT_TIME,
                 INSERT_USER,
                 INSERT_CLIENT,
                 UPDATE_TIME,
                 UPDATE_USER,
                 UPDATE_CLIENT,
                 UPDATE_COUNT
          FROM TF_MENU 
          WHERE IS_CUSTOM = 0 AND NOT EXISTS (SELECT ID FROM TF_MENU M WHERE M.IS_CUSTOM = 1 AND TF_MENU.ID = M.FACTORY_ID)
         )
