          FROM TF_LINKS
          WHERE IS_CUSTOM = 0 AND NOT EXISTS (SELECT ID FROM TF_LINKS L WHERE L.IS_CUSTOM = 1 AND TF_LINKS.ID = L.FACTORY_ID)
