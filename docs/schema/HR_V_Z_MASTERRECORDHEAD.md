		MIN(HR_TRANSPOSEDENTITY.EFFECTIVETO) AS EffectiveEndDate, 
		MAX(HR_TRANSPOSEDENTITY.CHANGEDATE) AS LastChangeDate
	FROM
		HR_TRANSPOSEDENTITY
	GROUP BY 
		HR_TRANSPOSEDENTITY.TASK_ID, HR_TRANSPOSEDENTITY.ADMINISTRATIVEEVENTCODE )
